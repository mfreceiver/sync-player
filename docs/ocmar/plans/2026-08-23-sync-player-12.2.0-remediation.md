# Sync Player 12.2.0 Remediation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use ocmar-subagent-driven-development (recommended) or ocmar-executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在单版本 12.2.0 内完成审计整改：修复 SMB/SFTP 取消误分类（CQ-001）、删除 dead `updateStarByPath` 链、把 `SyncEngine` 按生命周期拆为 4 个协作类、把 `BrowserViewModel` 的 Flow 组合与偏好持久化移入 `BrowserContentCoordinator`、清零项目源码 compiler/lint 告警，并走完整发版流程（依据已批准设计 `docs/ocmar/specs/2026-08-23-sync-player-12.2.0-remediation-design.md`，含 2026-08-23 Oracle 补正）。

**Architecture:** 方案 B（分阶段单版本）：transport CE-first 先行（TDD 红/绿/守卫三集），引擎按 Pending→Delete→Scanner→Download 依赖序拆分（每步迁移后既有 4 组集成测试零语义修改通过），Browser 协调器独立 lane（含 `UserPreferencesRepository` 依赖与 hydration 迁移），lint 分机械/行为两批（行为项以 suppression+理由为主，不改运行时语义）。

**Tech Stack:** Kotlin + Coroutines（JVM target 17）、Hilt、sshj、jcifs-ng、MockK（JVM 端 mock final 类——`SyncEngineTest` 已 mock 非 open 的 `VideoRepository` 为证）、androidx compose-bom 2025.12.00（compose-ui 1.10.0，含 `androidx.compose.ui.platform.LocalResources`）、JUnit4 + `kotlinx-coroutines-test`。单测配置 `unitTests.isReturnDefaultValues = true`（`app/build.gradle.kts:110`）——`android.util.Log` 在 JVM 测试返回默认值，无需 mock。

// app/src/test/java/cn/vectory/syncplayer/sync/sftp/SftpTransportCancellationTest.kt
package cn.vectory.syncplayer.sync.sftp
// imports：android.content/NET 系列、data.model.SyncProtocol/SyncSource、io.mockk.*、
//   kotlinx.coroutines.{CancellationException,CompletableDeferred,CoroutineScope,Job,launch}、
//   kotlinx.coroutines.test.runTest、net.schmizz.sshj.{SSHClient,sftp.*}、org.junit.Assert.*、
//   java.io.IOException、java.util.concurrent.{CountDownLatch,TimeUnit}、java.util.concurrent.atomic.AtomicInteger
/** CQ-001 回归：阻塞期间协程取消、阻塞随后以普通 IOException 完成时，操作必须以 CE 终结。 */
class SftpTransportCancellationTest {
    private val context = mockk<Context>()
    private val sftp = mockk<SFTPClient>()
    private val ssh = mockk<SSHClient>(relaxed = true) {
        every { newSFTPClient() } returns sftp
        every { disconnect() } just Runs
    }
    private val sut = SftpFileClient(context, sshClientFactory = { ssh })
    private val source = SyncSource(name = "s", protocol = SyncProtocol.SFTP, host = "host", port = 22, username = "u", password = "p")
    private fun stubNetworkOk() {   // NetworkUtils.checkNetworkAvailable 的 JVM stub（activeNetwork+INTERNET 能力）
        val cm = mockk<android.net.ConnectivityManager>(); val net = mockk<android.net.Network>()
        val caps = mockk<android.net.NetworkCapabilities>()
        every { context.getSystemService(Context.CONNECTIVITY_SERVICE) } returns cm
        every { cm.activeNetwork } returns net
        every { cm.getNetworkCapabilities(net) } returns caps
        every { caps.hasCapability(android.net.NetworkCapabilities.NET_CAPABILITY_INTERNET) } returns true
    }
    /** 阻塞开始→取消 parent→放行→返回终结异常（null=正常返回）。latch 超时统一 10s。 */
    private suspend fun awaitTerminalAfterCancel(
        started: CountDownLatch, release: CountDownLatch, block: suspend () -> Unit
    ): Throwable? {
        val outcome = CompletableDeferred<Throwable?>(); val parentJob = Job()
        CoroutineScope(parentJob).launch {
            try { block(); outcome.complete(null) } catch (t: Throwable) { outcome.complete(t) }
        }
        assertTrue("blocking op never started", started.await(10, TimeUnit.SECONDS))
        parentJob.cancel(); release.countDown()
        return outcome.await()
    }
    private fun assertCancelled(t: Throwable?) =
        assertTrue("expected CancellationException but was $t", t is CancellationException)
    private fun blockThenThrow(): Nothing = throw IOException("simulated transport timeout")
    private fun dirEntry(name: String): RemoteResourceInfo = mockk {
        every { this@mockk.name } returns name; every { isDirectory } returns true
        every { isRegularFile } returns false
        every { attributes } returns mockk { every { size } returns 0L; every { mtime } returns 0L }
    }
    // ---- 红集（修复前 FAIL：包装型边界）----
    @Test fun `connect red - cancelled connect completing with IOException must terminate with CE`() = runTest {
        stubNetworkOk()
        val started = CountDownLatch(1); val release = CountDownLatch(1)
        every { ssh.connect(any<String>(), any<Int>()) } answers {
            started.countDown(); release.await(10, TimeUnit.SECONDS); blockThenThrow()
        }
        assertCancelled(awaitTerminalAfterCancel(started, release) { sut.connect(source, "/remote") })
    }
    // ---- 守卫集（修复前后均 PASS：withContext 正常完成检查已恢复 CE）----
    @Test fun `listDirectories guard - cancelled ls still terminates with CE`() = runTest {
        stubNetworkOk(); sut.connect(source, "/remote")
        val started = CountDownLatch(1); val release = CountDownLatch(1)
        every { sftp.ls(any<String>()) } answers {
            started.countDown(); release.await(10, TimeUnit.SECONDS); blockThenThrow()
        }
        assertCancelled(awaitTerminalAfterCancel(started, release) { sut.listDirectories("/remote") })
    }
    @Test fun `listFiles guard - cancelled traversal does not scan deeper directories`() = runTest {
        stubNetworkOk(); sut.connect(source, "/remote")
        val rootScanned = CountDownLatch(1); val gate = CountDownLatch(1); val deeperCalls = AtomicInteger(0)
        every { sftp.ls(any<String>()) } answers {
            if (firstArg<String>() == "/remote") { rootScanned.countDown(); gate.await(10, TimeUnit.SECONDS); listOf(dirEntry("a")) }
            else { deeperCalls.incrementAndGet(); emptyList<RemoteResourceInfo>() }
        }
        assertCancelled(awaitTerminalAfterCancel(rootScanned, gate) { sut.listFiles("/remote") })
        assertEquals("cancelled traversal must not scan deeper directories", 0, deeperCalls.get())
    }
    @Test fun `downloadTo red - cancelled stat must terminate with CE not SyncException`() = runTest {
        stubNetworkOk(); sut.connect(source, "/remote")
        val started = CountDownLatch(1); val release = CountDownLatch(1)
        every { sftp.stat(any<String>()) } answers {
            started.countDown(); release.await(10, TimeUnit.SECONDS); blockThenThrow()
        }
        assertCancelled(awaitTerminalAfterCancel(started, release) {
            sut.downloadTo("a.mp4", java.io.ByteArrayOutputStream(), null)
        })
    }
    @Test fun `deleteFile red - cancelled rm must terminate with CE not SyncException`() = runTest {
        stubNetworkOk(); sut.connect(source, "/remote")
        val started = CountDownLatch(1); val release = CountDownLatch(1)
        every { sftp.rm(any<String>()) } answers {
            started.countDown(); release.await(10, TimeUnit.SECONDS); blockThenThrow()
        }
        assertCancelled(awaitTerminalAfterCancel(started, release) { sut.deleteFile("a.mp4") })
    }
    // 未取消 IOException 降级保留
    @Test fun `listDirectories guard - uncancelled IOException still degrades to empty list`() = runTest {
        stubNetworkOk(); sut.connect(source, "/remote")
        every { sftp.ls(any<String>()) } throws IOException("plain io error")
        assertTrue(sut.listDirectories("/remote").isEmpty())
    }
}
